cmake_minimum_required(VERSION 3.16)
project(performance_lab LANGUAGES CXX)

option(PERFLAB_NATIVE "Tune GCC/Clang builds for this machine (not portable)" OFF)
option(PERFLAB_AVX2 "Require AVX2 in all three variants" OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

function(add_image_variant target variant)
    add_executable(${target} src/main.cpp src/kernels.cpp)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE PERFLAB_VARIANT="${variant}")
    # Intentionally omit LTO: stage boundaries and three call sites remain visible.
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc /Zi /Oy-)
        target_link_options(${target} PRIVATE /DEBUG)
        if(variant STREQUAL "O0")
            target_compile_options(${target} PRIVATE /Od /Ob0)
        else()
            target_compile_options(${target} PRIVATE /O2)
            if(variant STREQUAL "O3-scalar")
                target_compile_options(${target} PRIVATE /Qvec-)
            endif()
        endif()
        if(PERFLAB_AVX2)
            target_compile_options(${target} PRIVATE /arch:AVX2)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -g -fno-omit-frame-pointer)
        if(variant STREQUAL "O0")
            target_compile_options(${target} PRIVATE -O0)
        else()
            target_compile_options(${target} PRIVATE -O3)
            if(variant STREQUAL "O3-scalar")
                if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
                    target_compile_options(${target} PRIVATE -fno-vectorize -fno-slp-vectorize)
                else()
                    target_compile_options(${target} PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize)
                endif()
            endif()
        endif()
        if(PERFLAB_NATIVE)
            target_compile_options(${target} PRIVATE -march=native)
        endif()
        if(PERFLAB_AVX2)
            target_compile_options(${target} PRIVATE -mavx2)
        endif()
    endif()
endfunction()

add_image_variant(image_o0 "O0")
add_image_variant(image_o3 "O3")
add_image_variant(image_scalar "O3-scalar")

enable_testing()
add_test(NAME image_o0_self_test COMMAND image_o0 --self-test)
add_test(NAME image_o3_self_test COMMAND image_o3 --self-test)
add_test(NAME image_scalar_self_test COMMAND image_scalar --self-test)
